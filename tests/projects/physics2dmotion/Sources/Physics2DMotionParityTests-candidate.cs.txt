using System;
using System.Collections.Generic;
using Xunit;

namespace UnityEngine.Tests
{
    public class MotionPhaseRecorder : MonoBehaviour
    {
        public Collision2D enter;
        void OnCollisionEnter2D(Collision2D collision) => enter = collision;
    }

    public class Physics2DMotionParityTests : IDisposable
    {
        readonly List<GameObject> objects = new List<GameObject>();
        public Physics2DMotionParityTests()
        {
            Physics2D.simulationMode = SimulationMode2D.Script;
            Physics2D.gravity = new Vector2(0, -9.81f);
            Physics2D.autoSyncTransforms = false;
            Physics2D.defaultContactOffset = .01f;
            Physics2D.velocityIterations = 8; Physics2D.positionIterations = 3;
        }
        public void Dispose()
        {
            foreach (var o in objects) Object.DestroyImmediate(o);
            Physics2D.Simulate(.02f);
        }
        GameObject Make(string name, Vector2 position)
        {
            var o = new GameObject(name); objects.Add(o); o.transform.position = position; return o;
        }
        static void Native(float expected, float actual) => Assert.InRange(actual,
            expected - .001f * Math.Max(1f, Math.Abs(expected)),
            expected + .001f * Math.Max(1f, Math.Abs(expected)));

        // Both independent 6000.0.71f1 CPU recordings report the same evaluated
        // geometry and incoming point velocity across these material combinations.
        [Theory]
        [InlineData(0f, 0f, 11.53656f)]
        [InlineData(0f, .5f, 14.4207f)]
        [InlineData(0f, 1f, 19.2276f)]
        [InlineData(.2f, 0f, 11.53656f)]
        [InlineData(.2f, .5f, 14.4207f)]
        [InlineData(.2f, 1f, 19.2276f)]
        [InlineData(.6f, 0f, 11.53656f)]
        [InlineData(.6f, .5f, 14.4207f)]
        [InlineData(.6f, 1f, 19.2276f)]
        public void CallbackAndGetContactsKeepEvaluatedGeometryWithSolvedImpulse(float friction, float restitution, float impulse)
        {
            var ground = Make("ground", new Vector2(0, -.5f)).AddComponent<BoxCollider2D>();
            ground.size = new Vector2(30, 1);
            ground.sharedMaterial = new PhysicsMaterial2D { friction = .3f, bounciness = .2f };
            var ball = Make("ball", new Vector2(0, 5));
            var collider = ball.AddComponent<CircleCollider2D>(); collider.radius = .5f;
            collider.sharedMaterial = new PhysicsMaterial2D { friction = friction, bounciness = restitution };
            var body = ball.AddComponent<Rigidbody2D>(); body.linearVelocity = new Vector2(1, 0);
            var recorder = ball.AddComponent<MotionPhaseRecorder>();
            for (int i = 0; i < 49; i++) Physics2D.Simulate(.02f);
            Assert.NotNull(recorder.enter);
            Assert.Equal(1, recorder.enter.contactCount);
            var contact = recorder.enter.GetContact(0);
            Native(.9599996f, contact.point.x); Native(-.05231124f, contact.point.y);
            Native(-.124622464f, contact.separation);
            Native(-1, contact.relativeVelocity.x); Native(9.4176f, contact.relativeVelocity.y);
            Native(impulse, contact.normalImpulse);
            Assert.True(body.position.y > .46f);
            var contacts = new ContactPoint2D[2]; Assert.Equal(1, body.GetContacts(contacts));
            Native(contact.point.x, contacts[0].point.x); Native(contact.point.y, contacts[0].point.y);
            Native(contact.separation, contacts[0].separation);
            Native(contact.relativeVelocity.y, contacts[0].relativeVelocity.y);
            Native(contact.normalImpulse, contacts[0].normalImpulse);
        }
    }
}
